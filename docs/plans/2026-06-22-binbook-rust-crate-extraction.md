# BinBook Rust Crate Extraction Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Extract the BinBook binary format parser from C firmware into a reusable `no_std` Rust crate, wire it into the SquidScript firmware, and verify on actual hardware.

**Architecture:** A standalone `no_std` Rust crate in `github.com/parasquid/binbook` implements parsing and decompression. SquidScript consumes it as a git submodule. `squidvm-ffi` uses it to replace the C `vm_runtime_binbook.c` callbacks. The firmware display backend loses its PackBits decompressor and receives decompressed pixels from the crate.

**Tech Stack:** Rust (`no_std`), `lz4_flex` (no_std), Zephyr C firmware, ESP32-C3 hardware

**Spec:** `docs/specs/2026-06-22-binbook-rust-crate-design.md`

```rust
use crate::Error;

pub(crate) const HEADER_SIZE: usize = 256;
pub(crate) const MAGIC: &[u8] = b"BINBOOK\0";

pub struct Header {
    pub file_size: u64,
    pub section_table_offset: u64,
    pub section_table_length: u32,
    pub section_table_entry_size: u16,
    pub section_count: u16,
    pub page_index_entry_size: u16,
    pub nav_index_entry_size: u16,
    pub page_data_offset: u64,
    pub page_data_length: u64,
}

fn read_le16(data: &[u8], offset: usize) -> u16 {
    u16::from_le_bytes([data[offset], data[offset + 1]])
}

fn read_le32(data: &[u8], offset: usize) -> u32 {
    u32::from_le_bytes([
        data[offset],
        data[offset + 1],
        data[offset + 2],
        data[offset + 3],
    ])
}

fn read_le64(data: &[u8], offset: usize) -> u64 {
    u64::from_le_bytes([
        data[offset],
        data[offset + 1],
        data[offset + 2],
        data[offset + 3],
        data[offset + 4],
        data[offset + 5],
        data[offset + 6],
        data[offset + 7],
    ])
}

pub(crate) fn parse_header(data: &[u8]) -> Result<Header, Error> {
    if data.len() < HEADER_SIZE {
        return Err(Error::InvalidHeader);
    }
    if &data[0..8] != MAGIC {
        return Err(Error::InvalidMagic);
    }
    let header_size = read_le16(data, 12);
    if header_size as usize != HEADER_SIZE {
        return Err(Error::UnsupportedVersion);
    }
    let h = Header {
        file_size: read_le64(data, 16),
        section_table_offset: read_le64(data, 24),
        section_table_length: read_le32(data, 32),
        section_table_entry_size: read_le16(data, 36),
        section_count: read_le16(data, 38),
        page_index_entry_size: read_le16(data, 40),
        nav_index_entry_size: read_le16(data, 42),
        page_data_offset: read_le64(data, 44),
        page_data_length: read_le64(data, 52),
    };
    if h.section_table_entry_size != 40
        || h.page_index_entry_size != 76
        || h.nav_index_entry_size != 48
        || h.section_table_offset < HEADER_SIZE as u64
        || h.section_count == 0
        || h.section_table_length < h.section_count as u32 * 40
    {
        return Err(Error::UnsupportedVersion);
    }
    Ok(h)
}

pub(crate) fn read_le16_pub(data: &[u8], offset: usize) -> u16 {
    read_le16(data, offset)
}

pub(crate) fn read_le32_pub(data: &[u8], offset: usize) -> u32 {
    read_le32(data, offset)
}

pub(crate) fn read_le64_pub(data: &[u8], offset: usize) -> u64 {
    read_le64(data, offset)
}
```

Create `<binbook-repo>/src/section.rs`:

```rust
use crate::Error;
use crate::header::{read_le16_pub as read_le16, read_le32_pub as read_le32, read_le64_pub as read_le64};

pub const SECTION_STRING_TABLE: u16 = 1;
pub const SECTION_PAGE_INDEX: u16 = 40;
pub const SECTION_NAV_INDEX: u16 = 41;
pub const SECTION_CHAPTER_INDEX: u16 = 43;
pub const SECTION_PAGE_DATA: u16 = 50;

pub const SECTION_ENTRY_SIZE: usize = 40;
pub const PAGE_INDEX_ENTRY_SIZE: usize = 76;
pub const NAV_INDEX_ENTRY_SIZE: usize = 48;
pub const CHAPTER_INDEX_ENTRY_SIZE: usize = 32;

#[derive(Debug, Clone, Default)]
pub struct Section {
    pub section_id: u16,
    pub offset: u64,
    pub length: u64,
    pub entry_size: u32,
    pub record_count: u32,
}

pub(crate) fn parse_section(bytes: &[u8]) -> Section {
    Section {
        section_id: read_le16(bytes, 0),
        offset: read_le64(bytes, 4),
        length: read_le64(bytes, 12),
        entry_size: read_le32(bytes, 20),
        record_count: read_le32(bytes, 24),
    }
}

pub(crate) struct RequiredSections {
    pub string_table: Section,
    pub page_index: Section,
    pub nav_index: Section,
    pub chapter_index: Section,
    pub page_data: Section,
}

pub(crate) fn find_sections(
    data: &[u8],
    section_table_offset: u64,
    section_count: u16,
    page_data_offset: u64,
    page_data_length: u64,
) -> Result<RequiredSections, Error> {
    let mut string_table = Section::default();
    let mut page_index = Section::default();
    let mut nav_index = Section::default();
    let mut chapter_index = Section::default();
    let mut page_data = Section::default();

    for i in 0..section_count {
        let off = section_table_offset as usize + i as usize * SECTION_ENTRY_SIZE;
        let end = off + SECTION_ENTRY_SIZE;
        if end > data.len() {
            return Err(Error::InvalidSection);
        }
        let section = parse_section(&data[off..end]);
        match section.section_id {
            SECTION_STRING_TABLE => string_table = section,
            SECTION_PAGE_INDEX => page_index = section,
            SECTION_NAV_INDEX => nav_index = section,
            SECTION_CHAPTER_INDEX => chapter_index = section,
            SECTION_PAGE_DATA => page_data = section,
            _ => {}
        }
    }

    if string_table.section_id != SECTION_STRING_TABLE {
        return Err(Error::MissingSection(SECTION_STRING_TABLE));
    }
    if page_index.section_id != SECTION_PAGE_INDEX {
        return Err(Error::MissingSection(SECTION_PAGE_INDEX));
    }
    if page_index.entry_size as usize != PAGE_INDEX_ENTRY_SIZE {
        return Err(Error::InvalidPageIndex);
    }
    if nav_index.section_id != SECTION_NAV_INDEX {
        return Err(Error::MissingSection(SECTION_NAV_INDEX));
    }
    if nav_index.entry_size as usize != NAV_INDEX_ENTRY_SIZE {
        return Err(Error::InvalidNavIndex);
    }
    if chapter_index.section_id != SECTION_CHAPTER_INDEX {
        return Err(Error::MissingSection(SECTION_CHAPTER_INDEX));
    }
    if chapter_index.entry_size as usize != CHAPTER_INDEX_ENTRY_SIZE {
        return Err(Error::InvalidChapterIndex);
    }
    if page_data.section_id != SECTION_PAGE_DATA {
        return Err(Error::MissingSection(SECTION_PAGE_DATA));
    }
    if page_index.record_count == 0 {
        return Err(Error::InvalidPageIndex);
    }
    if page_data.offset != page_data_offset || page_data.length != page_data_length {
        return Err(Error::InvalidSection);
    }

    Ok(RequiredSections {
        string_table,
        page_index,
        nav_index,
        chapter_index,
        page_data,
    })
}
```

```rust
use crate::Error;

/// Read a string from the string table.
///
/// Returns the byte slice for the string (not guaranteed UTF-8).
pub(crate) fn get_string<'a>(
    data: &'a [u8],
    string_table_offset: u64,
    string_table_length: u32,
    str_offset: u32,
    str_len: u32,
) -> Result<&'a [u8], Error> {
    if str_len == 0 {
        return Ok(b"");
    }
    if (str_offset as u64) + (str_len as u64) > string_table_length as u64 {
        return Err(Error::InvalidStringRef);
    }
    let abs = string_table_offset as usize + str_offset as usize;
    let end = abs + str_len as usize;
    if end > data.len() {
        return Err(Error::InvalidStringRef);
    }
    Ok(&data[abs..end])
}
```

```rust
mod error;
mod header;

pub struct BinBook<'a> {

    header: Header,
    sections: RequiredSections,

        self.sections.page_index.record_count
    }

    pub fn nav_count(&self) -> u32 {
        self.sections.nav_index.record_count
    }

    pub fn chapter_count(&self) -> u32 {
        self.sections.chapter_index.record_count
    }
}
```

```rust
use crate::Error;
use crate::header::{read_le16_pub as read_le16, read_le32_pub as read_le32};
use crate::section::CHAPTER_INDEX_ENTRY_SIZE;
use crate::string_table;

#[derive(Debug)]
pub struct ChapterEntry<'a> {
    pub index: u32,
    pub title: &'a [u8],
    pub page_index: u32,
    pub level: u16,
    pub entry_type: u16,
}

pub(crate) fn parse_chapter_entry<'a>(
    data: &'a [u8],
    chapter_index_offset: u64,
    index: u32,
    chapter_index_entry_size: u16,
    string_table_offset: u64,
    string_table_length: u32,
    page_count: u32,
) -> Result<ChapterEntry<'a>, Error> {
    let off = chapter_index_offset as usize + index as usize * chapter_index_entry_size as usize;
    let end = off + CHAPTER_INDEX_ENTRY_SIZE;
    if end > data.len() {
        return Err(Error::ChapterOutOfRange);
    }
    let bytes = &data[off..end];
    let chapter_index_raw = read_le32(bytes, 0);
    if chapter_index_raw != index {
        return Err(Error::InvalidChapterIndex);
    }
    let title_offset = read_le32(bytes, 8);
    let title_len = read_le32(bytes, 12);
    let pi = read_le32(bytes, 16);
    if pi >= page_count {
        return Err(Error::InvalidChapterIndex);
    }
    let entry_type = read_le16(bytes, 22);
    if entry_type != 3 && entry_type != 4 {
        return Err(Error::InvalidChapterIndex);
    }
    let title = string_table::get_string(data, string_table_offset, string_table_length, title_offset, title_len)?;
    Ok(ChapterEntry {
        index: chapter_index_raw,
        title,
        page_index: pi,
        level: read_le16(bytes, 20),
        entry_type,
    })
}

pub struct ChapterIter<'a> {
    data: &'a [u8],
    chapter_index_offset: u64,
    chapter_index_entry_size: u16,
    string_table_offset: u64,
    string_table_length: u32,
    page_count: u32,
    next_index: u32,
    remaining: u32,
}

impl<'a> ChapterIter<'a> {
    pub(crate) fn new(
        data: &'a [u8],
        chapter_index_offset: u64,
        chapter_index_entry_size: u16,
        string_table_offset: u64,
        string_table_length: u32,
        page_count: u32,
        offset: u32,
        limit: u32,
        total: u32,
    ) -> Self {
        let remaining = limit.min(total.saturating_sub(offset));
        Self {
            data,
            chapter_index_offset,
            chapter_index_entry_size,
            string_table_offset,
            string_table_length,
            page_count,
            next_index: offset,
            remaining,
        }
    }
}

impl<'a> Iterator for ChapterIter<'a> {
    type Item = Result<ChapterEntry<'a>, Error>;

    fn next(&mut self) -> Option<Self::Item> {
        if self.remaining == 0 {
            return None;
        }
        let result = parse_chapter_entry(
            self.data,
            self.chapter_index_offset,
            self.next_index,
            self.chapter_index_entry_size,
            self.string_table_offset,
            self.string_table_length,
            self.page_count,
        );
        self.next_index += 1;
        self.remaining -= 1;
        Some(result)
    }
}
```

```rust
use crate::page_index::{COMPRESSION_NONE, COMPRESSION_RLE_PACKBITS, COMPRESSION_LZ4};
use crate::PageRef;

/// Decompress a page into a caller-provided buffer.
///
/// Dispatches on page.compression_method:
///   0 (NONE) — memcpy
///   1 (RLE_PACKBITS) — PackBits decode
///   2 (LZ4) — lz4_flex decompress
pub fn decompress_page(page: &PageRef<'_>, out: &mut [u8]) -> Result<(), Error> {
    if out.len() < page.uncompressed_size {
        return Err(Error::OutputBufferTooSmall);
    }
    match page.info.compression_method {
        COMPRESSION_NONE => {
            if page.compressed_data.len() > out.len() {
                return Err(Error::OutputBufferTooSmall);
            }
            out[..page.compressed_data.len()].copy_from_slice(page.compressed_data);
            Ok(())
        }
        COMPRESSION_RLE_PACKBITS => {
            super::rle::decompress_packbits(page.compressed_data, out, page.uncompressed_size)
        }
        COMPRESSION_LZ4 => {
            super::lz4_decompress::decompress_lz4(page.compressed_data, out, page.uncompressed_size)
        }
        other => Err(Error::UnsupportedCompression(other)),
    }
}
```

```rust
use crate::Error;

/// Decompress PackBits (RLE) encoded data.
///
/// PackBits format: each packet starts with a signed byte n:
///   - 0..=127: copy the next n+1 bytes literally
///   - -127..=-1: repeat the next byte -n+1 times
///   - -128: no-op
pub fn decompress_packbits(input: &[u8], out: &mut [u8], expected_size: usize) -> Result<(), Error> {
    let mut in_pos = 0;
    let mut out_pos = 0;

    while in_pos < input.len() && out_pos < expected_size {
        let control = input[in_pos] as i8;
        in_pos += 1;

        if control >= 0 {
            // Literal run: copy control + 1 bytes
            let count = (control as usize) + 1;
            if in_pos + count > input.len() || out_pos + count > expected_size {
                return Err(Error::DecompressFailed);
            }
            out[out_pos..out_pos + count].copy_from_slice(&input[in_pos..in_pos + count]);
            in_pos += count;
            out_pos += count;
        } else if control != -128 {
            // Repeat run: repeat next byte -control + 1 times
            let count = (-(control as i16) + 1) as usize;
            if in_pos >= input.len() || out_pos + count > expected_size {
                return Err(Error::DecompressFailed);
            }
            let value = input[in_pos];
            in_pos += 1;
            for i in 0..count {
                out[out_pos + i] = value;
            }
            out_pos += count;
        }
        // control == -128: no-op, skip
    }

    if out_pos != expected_size {
        return Err(Error::DecompressFailed);
    }
    Ok(())
}
```

```rust
/// Decompress LZ4 block data into the output buffer.
pub fn decompress_lz4(input: &[u8], out: &mut [u8], expected_size: usize) -> Result<(), Error> {
    let decompressed = lz4_flex::decompress(input, expected_size)
        .map_err(|_| Error::DecompressFailed)?;
    if decompressed.len() != expected_size {
        return Err(Error::DecompressFailed);
    }
    if out.len() < expected_size {
        return Err(Error::OutputBufferTooSmall);
    }
    out[..expected_size].copy_from_slice(&decompressed);
    Ok(())
}
```
